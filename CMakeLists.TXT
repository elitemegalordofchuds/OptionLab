cmake_minimum_required(VERSION 3.20)
project(quant_engine VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    if(MSVC) 
        add_compile_options(/O2 /arch:AVX2 /W4)
    else() 
        add_compile_options(-O3 -mavx2 -mfma -Wall -Wextra)
    endif()
elseif(MSVC)
    add_compile_options(/O2 /arch:AVX2 /W4)
else()
    add_compile_options(-O3 -mavx2 -mfma -Wall -Wextra)
endif()

include_directories(include)

add_executable(option_pricer src/main.cpp "include/RNG.hpp"  "include/Types.hpp" "src/Monte-carlo.cpp" "src/Grid-solve.cpp" "src/Black-scholes.cpp" "include/input.hpp")
